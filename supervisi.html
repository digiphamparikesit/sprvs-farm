import { supabase } from './supabase-client.js'; import { showAlert } from './utils.js'; document.addEventListener('DOMContentLoaded', async () => { const tbody = document.getElementById('tbody-checklist'); const selectRuangan = document.getElementById('ruangan'); const form = document.getElementById('form-supervisi'); let semuaPertanyaan = []; // 1. LOAD RUANGAN async function loadRuangan() { const { data, error } = await supabase .from('ruangan') .select('*') .order('nama_ruangan'); if (error) { console.error('Gagal memuat ruangan:', error); showAlert('Gagal memuat daftar ruangan: ' + error.message, 'error'); return; } selectRuangan.innerHTML = ''; data.forEach(r => { const opt = document.createElement('option'); opt.value = r.id; opt.textContent = r.nama_ruangan; opt.dataset.nama = r.nama_ruangan; selectRuangan.appendChild(opt); }); } // 2. LOAD SEMUA MASTER PERTANYAAN async function loadMasterPertanyaan() { const { data, error } = await supabase .from('master_pertanyaan') .select('*') .order('no_urut', { ascending: true }) .order('id', { ascending: true }); if (error) { console.error('Gagal memuat pertanyaan:', error); showAlert('Gagal memuat pertanyaan: ' + error.message, 'error'); return; } semuaPertanyaan = data || []; } // 3. RENDER TABEL function renderPertanyaan(filterRuangan = '') { if (!filterRuangan) { tbody.innerHTML = 'Silakan pilih Unit/Ruangan terlebih dahulu untuk menampilkan pertanyaan.'; return; } const pertanyaanTersaring = semuaPertanyaan.filter(q => { if (!q.default_ruangan) return true; const ruanganDB = q.default_ruangan.toLowerCase(); const ruanganDipilih = filterRuangan.toLowerCase(); return ruanganDB.includes(ruanganDipilih) || ruanganDipilih.includes(ruanganDB); }); if (pertanyaanTersaring.length === 0) { tbody.innerHTML = 'Tidak ada pertanyaan spesifik untuk ruangan ini.'; return; } let html = ''; let lastKelompok = ''; pertanyaanTersaring.forEach((q) => { if (q.kelompok !== lastKelompok) { html += `${q.kelompok}`; lastKelompok = q.kelompok; } html += ` ${q.no_urut} ${q.objek_pemeriksaan} ${q.acuan} ${q.pengamatan_dilapangan} ${q.kategori} ${q.default_ruangan || '-'} `; }); tbody.innerHTML = html; } // 4. EVENT DELEGATION RADIO S/TS tbody.addEventListener('change', (e) => { if (e.target.type === 'radio' && e.target.name.startsWith('status_')) { const row = e.target.closest('tr'); const catatanInput = row.querySelector('.input-catatan'); if (e.target.value === 'TS') { catatanInput.disabled = false; catatanInput.required = true; catatanInput.focus(); row.style.backgroundColor = '#fff3cd'; } else { catatanInput.disabled = true; catatanInput.required = false; catatanInput.value = ''; row.style.backgroundColor = ''; } } }); // 5. DROPDOWN RUANGAN DIUBAH selectRuangan.addEventListener('change', (e) => { const selectedOption = e.target.options[e.target.selectedIndex]; const namaRuangan = selectedOption.dataset.nama || ''; renderPertanyaan(namaRuangan); }); // 6. HANDLE SUBMIT FORM form.addEventListener('submit', async (e) => { e.preventDefault(); const btnSubmit = document.getElementById('btn-submit'); const originalText = btnSubmit.textContent; btnSubmit.disabled = true; btnSubmit.textContent = 'Menyimpan...'; const rows = document.querySelectorAll('.pertanyaan-row'); if (rows.length === 0) { showAlert('Tidak ada pertanyaan yang bisa disimpan. Pilih ruangan terlebih dahulu.', 'warning'); btnSubmit.disabled = false; btnSubmit.textContent = originalText; return; } const ruanganId = document.getElementById('ruangan').value; if (!ruanganId) { showAlert('Pilih Unit/Ruangan terlebih dahulu!', 'warning'); btnSubmit.disabled = false; btnSubmit.textContent = originalText; return; } let isValid = true; let firstInvalidRow = null; const jawabanArray = []; rows.forEach(row => { const qId = row.dataset.id; const selectedRadio = row.querySelector(`input[name="status_${qId}"]:checked`); // PERBAIKAN: Gunakan backtick (`) bukan kutip biasa (') const catatanInput = row.querySelector(`textarea[name="catatan_${qId}"]`); if (!selectedRadio) { isValid = false; row.style.backgroundColor = '#f8d7da'; if (!firstInvalidRow) firstInvalidRow = row; } else if (selectedRadio.value === 'TS' && (!catatanInput || catatanInput.value.trim() === '')) { isValid = false; row.style.backgroundColor = '#f8d7da'; if (!firstInvalidRow) firstInvalidRow = row; if (catatanInput) catatanInput.focus(); } else { row.style.backgroundColor = ''; jawabanArray.push({ pertanyaan_id: qId, status: selectedRadio.value, catatan_temuan: selectedRadio.value === 'TS' ? catatanInput.value.trim() : null }); } }); if (!isValid) { showAlert('Harap jawab semua pertanyaan (S/TS) dan isi catatan jika memilih TS.', 'warning'); if (firstInvalidRow) firstInvalidRow.scrollIntoView({ behavior: 'smooth', block: 'center' }); btnSubmit.disabled = false; btnSubmit.textContent = originalText; return; } const headerData = { tanggal: document.getElementById('tanggal').value, jam: document.getElementById('jam').value, ruangan_id: ruanganId, kategori_risiko: document.getElementById('kategori_risiko').value, apoteker_supervisor_nama: document.getElementById('supervisor').value, penanggung_jawab_nama: document.getElementById('pj').value, metode: document.getElementById('metode').value, sampel: document.getElementById('sampel').value }; try { const { data: headerInserted, error: errHeader } = await supabase .from('supervisi') .insert([headerData]) .select() .single(); if (errHeader) throw new Error('Header: ' + errHeader.message); const detailData = jawabanArray.map(j => ({ supervisi_id: headerInserted.id, ...j })); const { error: errDetail } = await supabase .from('supervisi_detail') .insert(detailData); if (errDetail) { await supabase.from('supervisi').delete().eq('id', headerInserted.id); throw new Error('Detail: ' + errDetail.message); } showAlert('Data supervisi berhasil disimpan!', 'success'); window.location.href = `verifikasi.html?id=${headerInserted.id}`; } catch (error) { console.error('DETAIL ERROR:', error); showAlert('Gagal menyimpan data: ' + (error.message || 'Terjadi kesalahan tidak diketahui'), 'error'); } finally { btnSubmit.disabled = false; btnSubmit.textContent = originalText; } }); // 7. INISIALISASI await loadRuangan(); await loadMasterPertanyaan(); renderPertanyaan(''); });
